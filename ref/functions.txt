    -- invite codes
      create or replace function generate_invite_code()
      returns text as $$
      declare
        chars text := 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; -- no 0,O,1,I
        result text := '';
        i int;
      begin
        for i in 1..6 loop
          result := result || substr(chars, floor(random() * length(chars) + 1)::int, 1);
        end loop;
        return result;
      end;
      $$ language plpgsql;


      create or replace function join_club_with_code(input_code text)
      returns json
      language plpgsql
      security definer
      set search_path = public
      as $$
      declare
        v_club_id uuid;
        v_role_id uuid;
        v_membership_id uuid; -- Used to check if insert happened
      begin
        -- 1. Look up Club ID from the codes table
        -- CHANGED: selecting 'id', not 'club_id'
        select id into v_club_id
        from club_codes
        where code = input_code;

        if v_club_id is null then
          return json_build_object('success', false, 'error', 'Invalid Invite Code');
        end if;

        -- 2. Get 'Learner' role ID
        select id into v_role_id
        from club_roles
        where name = 'Learner';

        if v_role_id is null then
          raise exception 'Default role not configured';
        end if;

        -- 3. Insert membership
        -- We use RETURNING id to see if the row was actually created
        insert into club_members (club_id, profile_id, role_id)
        values (v_club_id, auth.uid(), v_role_id)
        on conflict (club_id, profile_id) do nothing
        returning id into v_membership_id;

        -- 4. Check if insert happened
        if v_membership_id is null then
          -- If null, it means ON CONFLICT triggered (already a member)
          return json_build_object('success', false, 'error', 'You are already a member of this club');
        end if;

        return json_build_object('success', true, 'club_id', v_club_id);
      end;
      $$;

      create or replace function get_club_preview(input_code text)
        returns json
        language plpgsql
        security definer -- Necessary to read the hidden 'club_codes' table
        set search_path = public
        as $$
        declare
          club_record record;
          member_count int;
        begin
          -- 1. Find club by joining the hidden codes table
          select c.id, c.name, c.description, c.created_at
          into club_record
          from club_codes cc
          join clubs c on c.id = cc.id
          where cc.code = input_code;
          
          if not found then
            return json_build_object('success', false, 'error', 'Invalid code');
          end if;
          
          -- 2. Get member count
          select count(*) into member_count
          from club_members
          where club_id = club_record.id;
          
          return json_build_object(
            'success', true,
            'id', club_record.id,
            'name', club_record.name,
            'description', club_record.description,
            'member_count', member_count,
            'created_at', club_record.created_at
          );
        end;
        $$;

      create or replace function handle_new_club_code()
        returns trigger 
        security definer
        as $$
        begin
          -- Loop to ensure uniqueness (rare edge case)
          loop
            begin
              insert into club_codes (id, code)
              values (new.id, generate_invite_code());
              exit; -- Exit loop if insert succeeds
            exception when unique_violation then
              -- If code exists, loop repeats and tries a new random code
              continue;
            end;
          end loop;
          return new;
        end;
        $$ language plpgsql;

      -- Attach the trigger
      create or replace trigger on_club_created
        after insert on clubs
        for each row execute function handle_new_club_code();

    -- auth check
      
      create or replace function auth_has_permission(p_club_id uuid, p_perm text)
      returns boolean
      language sql
      stable 
      security definer -- optional, but ensures it works even if roles table is locked down
      set search_path = public
      as $$
        select exists (
          select 1
          from club_members m
          join club_roles r on r.id = m.role_id
          where m.club_id = p_club_id
            and m.profile_id = auth.uid() -- Note: using 'profile_id' as per your previous schema
            and p_perm = ANY(r.permissions) -- The Array Check
        );
      $$;

    -- kick
      create or replace function kick_member(p_club_member_id uuid)
      returns json
      language plpgsql
      security definer
      set search_path = public
      as $$
      declare
        v_club_id uuid;
        v_target_order int;
        v_my_order int;
      begin
        -- -----------------------------------------------------
        -- 1. Get Target Info (Club & Rank)
        -- We need the club_id to know where we are check permissions
        -- -----------------------------------------------------
        select 
          cm.club_id, 
          r.order
        into 
          v_club_id, 
          v_target_order
        from club_members cm
        join club_roles r on r.id = cm.role_id
        where cm.id = p_club_member_id
          and cm.left_at is null;

        if v_club_id is null then
          return json_build_object('success', false, 'error', 'Member not found or already removed');
        end if;

        -- -----------------------------------------------------
        -- 2. Capability Check
        -- Now that we know the club_id, we check if YOU can delete members there
        -- -----------------------------------------------------
        if not auth_has_permission(v_club_id, 'club_member:kick') then
          return json_build_object('success', false, 'error', 'Permission denied');
        end if;

        -- -----------------------------------------------------
        -- 3. Get My Rank (The Kicker)
        -- Who am I in this specific club?
        -- -----------------------------------------------------
        select r.order into v_my_order
        from club_members cm
        join club_roles r on r.id = cm.role_id
        where cm.club_id = v_club_id
          and cm.profile_id = auth.uid()
          and cm.left_at is null;

        if v_my_order is null then
          return json_build_object('success', false, 'error', 'You are not active in this club');
        end if;

        -- -----------------------------------------------------
        -- 4. Hierarchy Check
        -- My Order must be lower (better) than Target Order
        -- -----------------------------------------------------
        if v_my_order >= v_target_order then
          return json_build_object(
            'success', false, 
            'error', 'You cannot remove a member with the same or more superior role.'
          );
        end if;

        -- -----------------------------------------------------
        -- 5. Execute Soft Delete
        -- -----------------------------------------------------
        update club_members
        set left_at = now()
        where id = p_club_member_id;

        return json_build_object('success', true);
      end;
      $$;


  -- projects
    create or replace function auth_can_manage_project(p_project_id uuid, p_perm text)
    returns boolean
    language sql
    stable
    security definer -- Vital for performance and preventing recursion
    set search_path = public
    as $$
      select 
        -- PATH 1: CLUB LEVEL (e.g. Club Guide)
        -- Checks if the project is linked to a club where I have permission
        exists (
          select 1
          from projects_clubs pc
          join club_members cm on cm.club_id = pc.club_id
          join club_roles r on r.id = cm.role_id
          where pc.project_id = p_project_id
            and cm.profile_id = auth.uid()
            and cm.left_at is null -- Must be an active club member
            and p_perm = ANY(r.permissions)
        )

        OR

        -- PATH 2: PROJECT LEVEL (Team & Followers)
        -- Checks if I am a direct member of the project with permission
        exists (
          select 1
          from project_members pm
          join project_roles pr on pr.id = pm.role_id
          where pm.project_id = p_project_id
            and pm.profile_id = auth.uid()
            and p_perm = ANY(pr.permissions)
        );
    $$;


    -- handle created_by
      create or replace function handle_new_project_creator()
        returns trigger
        security definer
        set search_path = public
        as $$
        declare
          v_role_id uuid;
          v_user_id uuid;
        begin
          -- 1. Get current user ID
          v_user_id := auth.uid();
          
          -- If created by a service role or anon (rare but possible), skip or handle
          if v_user_id is null then
            return new; 
          end if;

          -- 2. Find the 'Owner' role ID
          select id into v_role_id
          from project_roles
          where name = 'Creator';

          if v_role_id is null then
            raise exception 'Default project owner role not found.';
          end if;

          -- 3. Add the creator (current user) as a member
          insert into project_members (project_id, profile_id, role_id)
          values (new.id, v_user_id, v_role_id);

          return new;
        end;
        $$ language plpgsql;

        drop trigger if exists on_project_created on projects;

      create trigger on_project_created
        after insert on projects
        for each row execute function handle_new_project_creator();

drop view clubs_view;
create or replace view clubs_view
with (security_invoker = true) as
select
  clubs.*,
  -- Get the readable name (e.g. 'Guide', 'Learner')
  r.name as role_name,
  -- Get the capabilities (e.g. ['project:create', 'club:edit'])
  r.permissions,
  auth_has_permission(clubs.id, 'session:read') as session_read,
  auth_has_permission(clubs.id, 'session:create') as session_create
from
  clubs
-- Join 1: Link User to Club
left join club_members m 
  on m.club_id = clubs.id 
  and m.profile_id = auth.uid()
-- Join 2: Link Membership to Role Details
left join club_roles r
  on m.role_id = r.id;

      on profiles

        -- Option A: It is my own profile
        id = auth.uid()
        
        OR

        -- Option B: This profile is an active member of a club 
        -- where I have 'club_member:read_active' permission
        exists (
          select 1 
          from club_members cm
          where cm.profile_id = profiles.id    -- Link profile to its membership
          and cm.left_at is null               -- Only show if they are active (matches your specific logic)
          and auth_has_permission(cm.club_id, 'club_member:read_active') -- Reuse your helper!
